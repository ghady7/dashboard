-- 1. SQL Window Analysis: Customer Churn & Order Retention
WITH CustomerOrderCohorts AS (
    SELECT 
        CustomerID, OrderDate, OrderTotal,
        DATEDIFF(day, LAG(OrderDate, 1) OVER (
            PARTITION BY CustomerID ORDER BY OrderDate
        ), OrderDate) AS DaysSincePriorOrder,
        ROW_NUMBER() OVER (
            PARTITION BY CustomerID ORDER BY OrderDate DESC
        ) AS RecencyRank
    FROM Fact_CustomerOrders
)
SELECT 
    CASE WHEN RecencyRank = 1 AND DaysSincePriorOrder > 90 
         THEN 'At Risk' ELSE 'Active' END AS ChurnStatus,
    COUNT(DISTINCT CustomerID) AS CohortCount,
    ROUND(AVG(OrderTotal), 2)  AS AvgCustomerValue
FROM CustomerOrderCohorts
GROUP BY CASE WHEN RecencyRank = 1 AND DaysSincePriorOrder > 90 
              THEN 'At Risk' ELSE 'Active' END;

-- 2. Power BI DAX: Dynamic Customer Churn Rate %
ChurnRate% = 
VAR ActiveBase = CALCULATE([Total Customers], DATEADD(Dim_Date[Date], -90, DAY))
VAR Lapsed     = CALCULATE([Lost Customers], DATESBETWEEN(Dim_Date[Date], [PeriodStart], [PeriodEnd]))
RETURN DIVIDE(Lapsed, ActiveBase, 0)
